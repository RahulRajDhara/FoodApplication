CREATE TABLE [dbo].[FoodCategory] (      [CategoryId] INT IDENTITY(1,1) NOT NULL,      [CategoryName] VARCHAR(100) NULL,      [Description] VARCHAR(250) NULL,      [ImagePath] VARCHAR(500) NULL,      [IsActive] INT NULL,      [CreatedOn] DATETIME NULL,      [UpdatedOn] DATETIME NULL  );

CREATE TABLE [dbo].[FoodItems] (      [FoodId] INT IDENTITY(1,1) NOT NULL,      [CategoryId] INT NULL,      [FoodName] VARCHAR(150) NULL,      [Description] VARCHAR(500) NULL,      [Price] DECIMAL(10,2) NULL,      [ImagePath] VARCHAR(500) NULL,      [IsAvailable] INT NOT NULL,      [CreatedOn] DATETIME NULL,      [UpdatedOn] DATETIME NULL  );
CREATE TABLE [dbo].[Users] (      [UserId] INT IDENTITY(1,1) NOT NULL,      [FullName] VARCHAR(100) NULL,      [Email] VARCHAR(150) NULL,      [Phone] VARCHAR(15) NULL,      [PasswordHash] VARCHAR(255) NULL,      [Role] VARCHAR(20) NOT NULL,      [IsActive] INT NOT NULL,      [CreatedOn] DATETIME NULL,      [UpdatedOn] DATETIME NULL  );
