create procedure [dbo].[GetallFoodCatagories]
@In_Catid int=0
as
begin
if(isnull(@In_Catid,0)=0)
begin
 SELECT
        CategoryID,
        CategoryName,
        ImagePath
    FROM FoodCategory
    ORDER BY CategoryName;
end
end
go

CREATE procedure [dbo].[GetallFooditems]
@In_Catid int=0
as
begin
if(isnull(@In_Catid,0)=0)
begin
select FI.FoodId,FI.CategoryId,FC.CategoryName, FI.FoodName, isnull(FI.Description,'')Description, FI.Price, isnull(FI.ImagePath,'')ImagePath
from FoodItems FI
INNER JOIN FoodCategory FC ON FC.CategoryId=FI.CategoryId
where IsAvailable=1
end
else
begin
select FI.FoodId,FI.CategoryId,FC.CategoryName, FI.FoodName, isnull(FI.Description,'')Description, FI.Price, isnull(FI.ImagePath,'')ImagePath
from FoodItems FI
INNER JOIN FoodCategory FC ON FC.CategoryId=FI.CategoryId
where FI.CategoryId=@In_Catid and  FI.IsAvailable=1
end
end
